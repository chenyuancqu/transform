let sessions = {};

async function initModels() {
  sessions.rover = await ort.InferenceSession.create('/models/rover.onnx');
  sessions.uav1 = await ort.InferenceSession.create('/models/uav_1.onnx');
  sessions.uav2 = await ort.InferenceSession.create('/models/uav_2.onnx');
  sessions.uav3 = await ort.InferenceSession.create('/models/uav_3.onnx');
  console.log("全套强化学习模型初始化成功！");
}

// 2. 泛型推理函数
async function predict(session, inputData, shape) {
  const tensor = new ort.Tensor('float32', Float32Array.from(inputData), shape);
  const results = await session.run({ observation: tensor });
  return results.action.data; // 返回 Float32Array 动作
}

// 3. 仿真主循环示例 Step
async function simulationStep(state) {
  // --- A. 母车推理 ---
  const roverObs = [
    state.rover.x,
    state.rover.y,
    state.unservedCenter.x - state.rover.x,
    state.unservedCenter.y - state.rover.y,
    state.remainOrdersCount / 8.0,
    state.uavs.reduce((a, b) => a + b.battery, 0) / 3.0
  ];
  const roverAction = await predict(sessions.rover, roverObs, [1, 6]);
  
  // 更新母车位置
  state.rover.x = Math.max(0, Math.min(1, state.rover.x + roverAction[0] * 0.03));
  state.rover.y = Math.max(0, Math.min(1, state.rover.y + roverAction[1] * 0.03));

  // --- B. 3架无人机推理 ---
  for (let i = 0; i < 3; i++) {
    const uav = state.uavs[i];
    const session = sessions[`uav${i + 1}`];
    
    // 计算最近的订单
    const nearestOrderRel = getNearestOrderRel(uav, state.unservedOrders);

    const uavObs = [
      uav.x, uav.y, uav.z,
      uav.battery,
      state.rover.x - uav.x,
      state.rover.y - uav.y,
      0.0 - uav.z,
      nearestOrderRel.x, nearestOrderRel.y, nearestOrderRel.z
    ];

    const uavAction = await predict(session, uavObs, [1, 10]);

    // 更新无人机坐标与电量
    uav.x = Math.max(0, Math.min(1, uav.x + uavAction[0] * 0.06));
    uav.y = Math.max(0, Math.min(1, uav.y + uavAction[1] * 0.06));
    uav.z = Math.max(0, Math.min(0.3, uav.z + uavAction[2] * 0.06));
    uav.battery -= 0.012;

    // 检查充电逻辑
    const distToRover = Math.hypot(uav.x - state.rover.x, uav.y - state.rover.y);
    if (distToRover < 0.08 && uav.z < 0.05) {
      uav.battery = 1.0;
    }
  }

  // --- C. 刷新渲染页面 ---
  renderScene(state);
}

算法项目地址

本算法建模的是“地面无人母车 + 空中多无人机（1母车 + N无人机）陆空协同末端揽投”场景。

在数字孪生（Digital Twin）前端建模时，建议将其设定为以下**三个真实商业场景之一**，并按对应的 3D 资产和特效进行绘制：

---

### 一、 推荐的三种真实业务场景

#### 1. 首选场景：城市高层楼宇/阳台无人机揽投（Urban Balcony Logistics）

* **背景**：快递车（母车）行驶在地面主干道上，配送/揽收点分布在道路两侧高层建筑的**不同高度（阳台、露台、楼顶停机坪）**。
* **协同逻辑**：
* **母车（Rover）**：作为移动控制中心与“充电充电桩”，根据客户订单分布情况，在地面道路上移动，停靠在订单最密集的楼栋下方。
* **无人机（UAV）**：从母车顶舱起飞，垂直爬升至不同高度的阳台/窗台完成包裹揽收（对应 $Z$ 轴坐标），电量不足时自动飞回正在行驶或停靠的母车顶舱快速补能。



#### 2. 备选场景 2：偏远山区/农郊农产品与急件揽收（Rural/Mountainous Area Pickup）

* **背景**：地形复杂或道路通达度差，农户/取货点散落分布在山丘不同海拔高度（对应 3D 空间取货点）。
* **协同逻辑**：母车沿着山脚/主干道行驶，无人机飞往山坡各取货点揽收高附加值农产品（如明前茶、鲜花等），再返回母车汇总。

#### 3. 备选场景 3：大型半封闭园区/厂区巡检与快递揽收（Industrial Park Logistics）

* **背景**：大型制造园区或物流园区，包含多栋厂房、露天堆场、高架仓储。
* **协同逻辑**：母车在园区道路环线巡航，无人机负责各厂房高层窗口、堆场内部的急件揽收与巡检。

---

### 二、 前端数字孪生（3D 场景）建模资产清单

为了配合算法效果，建议前端准备以下 3D 模型与UI元素：

#### 1. 实体 3D 模型资产（Models）

* **地面母车（Rover）**：1 辆带顶部降落平台/舱门开合的**无人驾驶快递车/移动基站车**。
* **多旋翼无人机（UAV）**：3 架带小型货舱或机械抓手的**小型物流无人机**（可加灯光区分编号 UAV-01, 02, 03）。
* **取货/送货目标（Orders）**：8 个带有 3D 坐标的**包裹/智能取货箱**（可悬浮在楼宇阳台或特定空间位置）。
* **环境地图**：1 块长宽 1:1 的三维网格地图（如 100m × 100m 园区，高度 30m）。

#### 2. 动态视觉特效（Visual Effects & Animation）

* **母车充电光环**：当无人机靠近母车（距离 $< 0.08$ 且高度近地面）时，母车顶部触发**绿色脉冲光圈/电量补给粒子特效**。
* **无人机轨迹线（Trail）**：为 3 架无人机绘制动态拖尾线（或不同颜色的虚线），展示其在三维空间中的飞行路径。
* **包裹挂载状态**：无人机触碰取货点（距离 $< 0.08$）时，取货点包裹消失，无人机底部挂载包裹模型或显示“已揽收”高亮文本。
* **电量警戒 UI**：无人机上方浮动血条/电量条（低于 20% 变红警告，飞回母车充电变绿）。

---

### 三、 算法行为在前台渲染时的“看点”

在数字孪生画面中，算法会呈现出以下**符合物理直觉的智能化动作**：

1. **母车“接应”行为**：母车不会停在原地，而是会自动驶向当前未完成订单的“重心位置”，作为无人机的移动前线补给站，缩短无人机往返飞行距离。
2. **无人机“梯次补能”**：当某架无人机电量偏低时，它会放弃远端任务，强制划出一条抛物线飞回母车降落充电；其他电量充足的无人机则继续在外揽件。
3. **空间三维避障/寻路**：无人机会根据 $Z$ 轴高度进行升降，完成对高处目标的精准接近与返回。
